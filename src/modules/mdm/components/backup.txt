// app/(docs)/layout.js
"use client";
import { Box, Container, CssBaseline,IconButton,TextField, InputAdornment,Menu,MenuItem,CircularProgress } from "@mui/material";
import DocsSidebar from "./../components/DocsSidebar";
import React, { useRef, useEffect } from "react";
import { useState } from "react";
import { DrawerContext } from "@/context/DrawerContext";
import { useDrawer } from "@/context/DrawerContext";
import YTXTypography from "../components/YTXTypography";
import MenuIcon from "@mui/icons-material/Menu";
import SearchIcon from "@mui/icons-material/Search";

// export const metadata = {
//   title: "Help Center | YourMDM",
// };

export default function DocsLayout({ children }) {
  const [mobileOpen, setMobileOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState("");
  const [searchResults, setSearchResults] = useState([]);
  const [isLoading, setIsLoading] = useState(false);
  const [anchorEl, setAnchorEl] = useState(null);
  const searchRef = useRef(null);
  const [searchIndex, setSearchIndex] = useState([]);
  const { onToggle } = useDrawer();
  const handleDrawerToggle = () => {
    setMobileOpen((prev) => !prev);
  };

  const handleClose = () => setAnchorEl(null);

  // Load search index on mount
  useEffect(() => {
    fetch("/search-index.json")
      .then((res) => res.json())   // ✅ return json here
      .then((data) => {
        console.log("Search index content:", data);  // ✅ print content
        setSearchIndex(data);
  
        const fuse = new Fuse(data, {
          keys: ["title", "body"],
          threshold: 0.3,
          includeScore: true,
          limit: 5,
        });
  
        // Optional: store fuse in state if needed later
        // setFuse(fuse);
      })
      .catch((error) => {
        console.error("Error loading search index:", error);
      });
  }, []);
  

  // Handle search on change
  useEffect(() => {
    if (searchQuery.length < 2) {
      setSearchResults([]);
      setAnchorEl(null);
      return;
    }

    setIsLoading(true);
    // Simulate delay for loader (remove in prod)
    const timer = setTimeout(() => {
      const fuse = new Fuse(searchIndex, {
        keys: ["title", "body"],
        threshold: 0.3,
        includeScore: true,
        limit: 5,
      });
      const results = fuse.search(searchQuery).map((result) => ({
        ...result.item,
        score: result.score,
        snippet: result.item.body.substring(0, 200) + "...", // 2-3 line snippet
      }));
      setSearchResults(results);
      setAnchorEl(searchRef.current); // Open dropdown
      setIsLoading(false);
    }, 300); // Debounce

    return () => clearTimeout(timer);
  }, [searchQuery, searchIndex]);
  return (
    <>
      <CssBaseline />
      <DrawerContext.Provider value={{ onToggle: handleDrawerToggle }}>
        <Box sx={{ display: "flex", minHeight: "100vh", bgcolor: "#0a0e17" }}>
          <DocsSidebar
            mobileOpen={mobileOpen}
            onDrawerToggle={handleDrawerToggle}
          />

          {/* Fixed Sticky Search - Always Visible at Viewport Top */}
          <Box
            sx={{
              position: "fixed",
              top: 0,
              left: 0,
              right: 0,
              zIndex: 100,
              mb:20,
              backdropFilter: "blur(20px)",
              borderBottom: "1px solid rgba(255,255,255,0.08)",
              py: { xs: 1.5, md: 2 }, // responsive vertical padding
              display: "flex",
              justifyContent: "center",
            }}
          >
            <Container
              maxWidth="xl"
              sx={{
                display: "flex",
                alignItems: "center", // Align hamburger with search
                justifyContent: "center",
                gap: 2, // Space between hamburger and search
                px: { xs: 2, md: 0 },
              }}
            >
              {/* Hamburger for mobile */}
              <IconButton
                onClick={handleDrawerToggle}
                sx={{
                  color: "white",
                  display: { xs: "flex", md: "none" }, // Show only on mobile
                  zIndex: 110,
                }}
              >
                <MenuIcon />
              </IconButton>
              {/* Live Search with Dropdown */}
              <TextField
                inputRef={searchRef}
                fullWidth
                placeholder="Search documentation..."
                variant="outlined"
                value={searchQuery}
                onChange={(e) => {
                  setSearchQuery(e.target.value);
                  setAnchorEl(searchRef.current); // keep dropdown aligned
                }}
                slotProps={{
                  input: {
                    startAdornment: (
                      <InputAdornment position="start">
                        <SearchIcon sx={{ color: "grey.500" }} />
                      </InputAdornment>
                    ),
                  },
                }}
                sx={{
                  width: "100%",
                  maxWidth: { xs: "100%", sm: 500, md: 600, lg: 700 }, // responsive width
                  "& .MuiOutlinedInput-root": {
                    height: { xs: 52, md: 42 }, // responsive height
                    bgcolor: "rgba(255,255,255,0.05)",
                    borderRadius: 4,
                    border: "1px solid rgba(255,255,255,0.15)",
                    fontSize: { xs: "1rem", md: "1.1rem" },
                    color: "white",
                    "&:hover": { borderColor: "#00f5ff" },
                    "&.Mui-focused": {
                      borderColor: "#00f5ff",
                      boxShadow: "0 0 0 4px rgba(0,245,255,0.25)",
                    },
                  },
                }}
              />

              {/* Dropdown Menu with Loader/Results */}
              <Menu
                anchorEl={searchRef.current} // 🔥 Perfect align with input
                open={Boolean(anchorEl) && searchResults.length > 0}
                onClose={handleClose}
                anchorOrigin={{ vertical: "bottom", horizontal: "center" }}
                transformOrigin={{ vertical: "top", horizontal: "center" }}
                PaperProps={{
                  sx: {
                    width: "100%",
                    maxWidth: { xs: "100%", sm: 500, md: 600, lg: 700 },
                    maxHeight: 400,
                    mt: 1,
                    bgcolor: "rgba(2, 8, 16, 0.98)",
                    border: "1px solid rgba(0,245,255,0.2)",
                    borderRadius: 2,
                  },
                }}
              >
                {isLoading ? (
                  <Box sx={{ p: 3, textAlign: "center" }}>
                    <CircularProgress size={24} sx={{ color: "#00f5ff" }} />
                    <YTXTypography
                      variant="body2"
                      color="grey.500"
                      sx={{ mt: 1 }}
                    >
                      Searching...
                    </YTXTypography>
                  </Box>
                ) : (
                  searchResults.map((result) => (
                    <MenuItem
                      key={result.id}
                      onClick={handleClose}
                      component={Link}
                      href={result.path}
                      sx={{ whiteSpace: "normal" }}
                    >
                      <Box sx={{ width: "100%" }}>
                        <YTXTypography
                          variant="subtitle1"
                          fontWeight={600}
                          color="white"
                        >
                          {result.title}
                        </YTXTypography>

                        <YTXTypography
                          variant="body2"
                          color="grey.400"
                          sx={{ lineHeight: 1.4, mt: 0.5 }}
                        >
                          {result.snippet}
                        </YTXTypography>

                        <YTXTypography variant="caption" color="grey.600">
                          {result.path}
                        </YTXTypography>
                      </Box>
                    </MenuItem>
                  ))
                )}
              </Menu>
            </Container>
          </Box>

          <Container maxWidth="lg" sx={{mt:5}}>{children}</Container>
        </Box>
      </DrawerContext.Provider>
    </>
  );
}
